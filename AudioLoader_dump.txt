	private IEnumerator WaitForBundleToLoad()
	{
		if (loadRequest == null || ((AsyncOperation)loadRequest).isDone)
		{
			yield break;
		}
		CreateTextPanel();
		string prevText = "";
		while (!((AsyncOperation)loadRequest).isDone)
		{
			yield return null;
			string text = WaitText + $"({Math.Round(((AsyncOperation)loadRequest).progress, 1) * 100.0}%)";
			if (text != prevText)
			{
				TextPanelText.text = text;
				prevText = text;
			}
		}
		Object.Destroy((Object)(object)TextCanvas);
	}
